import 'dart:io';
import 'dart:ui' as ui;
import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter/services.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';
import 'package:qr_flutter/qr_flutter.dart';
import 'package:paskluis_v1/main.dart' as app;
import 'package:paskluis_v1/shared/widgets/main_tab_swipe_region.dart';

void main() {
  final binding = IntegrationTestWidgetsFlutterBinding.ensureInitialized();
  final metrics = <Map<String, Object?>>[];
  final captureKey = GlobalKey();
  Widget captureApp(Widget screen) => RepaintBoundary(
    key: captureKey,
    child: app.testApp(screen),
  );
  setUpAll(() async { await app.initializeFixtures(); });
  Future<void> ready(WidgetTester tester) async {
    await tester.pump();
    await Future<void>.delayed(const Duration(milliseconds: 900));
    // Match the test render surface to the actual native window after changes.
    // Keep this in the harness; never substitute a hypothetical device size.
    await binding.setSurfaceSize(tester.view.physicalSize / tester.view.devicePixelRatio);
    await tester.pump(const Duration(milliseconds: 300));
  }
  Future<void> shot(WidgetTester tester, String name) async {
    await ready(tester);
    expect(tester.takeException(), isNull);
    final view = tester.view;
    final context = tester.element(find.byType(Scaffold).first);
    final size = MediaQuery.sizeOf(context);
    final actualWindow = view.physicalSize / view.devicePixelRatio;
    final renderedSize = tester.getSize(find.byType(Scaffold).first);
    expect(renderedSize.width, closeTo(actualWindow.width, 0.1), reason: 'Rendered width matches native window');
    expect(renderedSize.height, closeTo(actualWindow.height, 0.1), reason: 'Rendered height matches native window');
    metrics.add({'name': name, 'physicalWidth': view.physicalSize.width,
      'physicalHeight': view.physicalSize.height, 'pixelRatio': view.devicePixelRatio,
      'logicalWidth': size.width, 'logicalHeight': size.height,
      'renderedWidth': renderedSize.width, 'renderedHeight': renderedSize.height,
      'padding': MediaQuery.paddingOf(context).toString(),
      'features': MediaQuery.of(context).displayFeatures.toString()});
    for (final qr in tester.widgetList<QrImageView>(find.byType(QrImageView))) {
      final rect = tester.getRect(find.byWidget(qr));
      expect(rect.width, closeTo(rect.height, 0.1), reason: 'Native QR remains square');
    }
    if (Platform.isAndroid) {
      await binding.convertFlutterSurfaceToImage();
      await tester.pump();
    }
    await binding.takeScreenshot(name);
    binding.reportData ??= {};
    binding.reportData!['nativeMetrics'] = metrics;
    // Independent Flutter-layer capture distinguishes a UIKit capture/rotation
    // problem from an incorrectly laid out app. Do not replace the native PNG.
    final boundary = captureKey.currentContext!.findRenderObject()! as RenderRepaintBoundary;
    final renderedImage = await boundary.toImage(pixelRatio: view.devicePixelRatio);
    final renderedBytes = await renderedImage.toByteData(format: ui.ImageByteFormat.png);
    (binding.reportData!['screenshots'] as List<dynamic>).add({
      'screenshotName': '$name-flutter',
      'bytes': renderedBytes!.buffer.asUint8List().toList(),
    });
    renderedImage.dispose();
  }
  for (final entry in <String, String>{
    'loyalty': 'Klantenkaarten', 'gifts': 'Cadeaukaarten', 'qr': 'QR-codes',
    'loyalty-pass': 'Klantenkaart openen', 'gift-pass': 'Cadeaukaart openen', 'qr-set': 'QR-set openen',
  }.entries) {
    testWidgets('native ${entry.key} fits actual device', (tester) async {
      await tester.pumpWidget(captureApp(app.testScreen(entry.value)));
      await shot(tester, entry.key);
      if (!entry.key.contains('pass') && entry.key != 'qr-set') {
        final context = tester.element(find.byType(Scaffold).first);
        final media = MediaQuery.of(context);
        final content = tester.getRect(find.byType(MainTabSwipeRegion));
        expect(content.left, greaterThanOrEqualTo(media.padding.left));
        expect(content.right, lessThanOrEqualTo(media.size.width - media.padding.right));
        final grid = tester.renderObject<RenderSliverGrid>(find.byType(SliverGrid).first);
        final layout = grid.gridDelegate.getLayout(grid.constraints) as SliverGridRegularTileLayout;
        expect(layout.crossAxisCount, greaterThanOrEqualTo(grid.constraints.crossAxisExtent >= 600 ? 3 : 1));
      }
      await tester.pumpWidget(const SizedBox.shrink());
    });
  }
  testWidgets('selected pass survives a native orientation request', (tester) async {
    await tester.pumpWidget(captureApp(app.testScreen('Klantenkaart openen')));
    await ready(tester);
    final controller = tester.widget<PageView>(find.byType(PageView)).controller!;
    controller.jumpToPage(3);
    await ready(tester);
    await SystemChrome.setPreferredOrientations([DeviceOrientation.landscapeLeft]);
    await shot(tester, 'loyalty-selected-landscape-request');
    expect(tester.widget<Scaffold>(find.byType(Scaffold).first).appBar, isNull,
      reason: 'A short native window uses the compact barcode layout');
    expect(controller.page, closeTo(3, 0.01));
    expect(find.text('Jumbo'), findsOneWidget);
    await SystemChrome.setPreferredOrientations([DeviceOrientation.portraitUp]);
    await shot(tester, 'loyalty-selected-portrait-request');
    expect(controller.page, closeTo(3, 0.01));
    expect(find.text('Jumbo'), findsOneWidget);
    await SystemChrome.setPreferredOrientations([]);
    await tester.pumpWidget(const SizedBox.shrink());
  });
}
